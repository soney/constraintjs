    Interface ArrayDiff<T>

    An edit script that turns one array into another. Applying removed, then added, then moved (each in order) to the old array produces the new one. See cjs.arrayDiff.

    interface ArrayDiff<T = any> {
        added: { item: T; to: number; to_item: T }[];
        index_changed: {
            from: number;
            from_item: T;
            item: T;
            to: number;
            to_item: T;
        }[];
        moved: {
            from: number
            | undefined;
            insert_at: number;
            item: T;
            move_from: number;
            to: number;
        }[];
        removed: { from: number; from_item: T }[];
    }
